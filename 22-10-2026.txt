        // Services Filter by Search Input
        function filterServices() {
            const query = document.getElementById('serviceSearch').value.toLowerCase();
            const cards = document.querySelectorAll('.service-card');

            cards.forEach(card => {
                const searchData = card.getAttribute('data-name').toLowerCase();
                if (searchData.includes(query)) {
                    card.style.display = 'flex';
                } else {
                    card.style.display = 'none';
                }
            });
        }

        // Services Filter by Category
        function filterCategory(cat) {
            const cards = document.querySelectorAll('.service-card');
            const buttons = document.querySelectorAll('.cat-btn');

            buttons.forEach(btn => {
                if (btn.getAttribute('data-cat') === cat) {
                    btn.classList.add('bg-navy-800', 'text-white');
                    btn.classList.remove('bg-white', 'text-slate-700');
                } else {
                    btn.classList.remove('bg-navy-800', 'text-white');
                    btn.classList.add('bg-white', 'text-slate-700');
                }
            });

            cards.forEach(card => {
                if (cat === 'all' || card.getAttribute('data-category') === cat) {
                    card.style.display = 'flex';
                } else {
                    card.style.display = 'none';
                }
            });
        }

        // Calculate Interactive Price Estimation
        function calculateTotal() {
            const checkboxes = document.querySelectorAll('.calc-checkbox');
            let original = 0;
            let selectedList = [];

            checkboxes.forEach(cb => {
                if (cb.checked) {
                    original += parseInt(cb.getAttribute('data-price'));
                    selectedList.push(cb.getAttribute('data-name'));
                }
            });

            const coupon = document.getElementById('couponCode').value.trim().toUpperCase();
            let discountRate = 0;

            if (coupon === 'FIRST20' && original > 0) {
                discountRate = 0.30; // 30% default discount for demo
            }

            const discount = Math.round(original * discountRate);
            const finalPrice = original - discount;

            document.getElementById('originalPrice').innerText = '₹' + original;
            document.getElementById('discountAmount').innerText = '- ₹' + discount;
            document.getElementById('finalPrice').innerText = '₹' + finalPrice;
        }

        // Send estimated price via WhatsApp
        function bookEstimatedServices() {
            const checkboxes = document.querySelectorAll('.calc-checkbox');
            let selectedList = [];

            checkboxes.forEach(cb => {
                if (cb.checked) {
                    selectedList.push(cb.getAttribute('data-name'));
                }
            });

            if (selectedList.length === 0) {
                alert('कृपया किमान एक सेवा निवडा!');
                return;
            }

            const finalPrice = document.getElementById('finalPrice').innerText;
            const text = `नमस्कार! मी कॅल्क्युलेटरद्वारे खालील सेवा निवडल्या आहेत:%0A- ${selectedList.join('%0A- ')}%0A%0Aअंदाजे किंमत: ${finalPrice} (FIRST20 सवलतीसह)`;
            window.open(`https://wa.me/919511807597?text=${text}`, '_blank');
        }
